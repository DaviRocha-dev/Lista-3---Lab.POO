public class Produto {
    private String nome;
    private double preco;

    public Produto(String nome, double preco) {
        this.nome = nome;
        this.preco = preco;
    }

    public double exibirPreco() {
        return preco;
    }
}

---------------------------------------------------

    public static void main(String[] args) {
